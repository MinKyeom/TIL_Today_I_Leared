        function playPopSound(freq = 440, type = 'sine') {
            if (!state.soundEnabled) return;
            try {
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = type;
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.15);
            } catch (e) { console.error(e); }
        }

        class DrumBall {
            constructor(number) {
                this.number = number;
                this.radius = 11;
                this.x = 140 + (Math.random() - 0.5) * 100;
                this.y = 140 + (Math.random() - 0.5) * 100;
                this.vx = (Math.random() - 0.5) * 8;
                this.vy = (Math.random() - 0.5) * 8;
                this.color = this.getColor(number);
            }

            getColor(num) {
                if (num <= 10) return '#fbc400';
                if (num <= 20) return '#69c8f2';
                if (num <= 30) return '#ff7272';
                if (num <= 40) return '#aaaaaa';
                return '#b0d840';
            }

            update(speedMultiplier) {
                this.x += this.vx * speedMultiplier;
                this.y += this.vy * speedMultiplier;

                // Circular Drum Collision Bounds
                const dist = Math.hypot(this.x - 140, this.y - 140);
                const maxR = 130 - this.radius;
                if (dist > maxR) {
                    const angle = Math.atan2(this.y - 140, this.x - 140);
                    this.x = 140 + Math.cos(angle) * maxR;
                    this.y = 140 + Math.sin(angle) * maxR;
                    this.vx = -this.vx * 0.9 + (Math.random() - 0.5) * 2;
                    this.vy = -this.vy * 0.9 + (Math.random() - 0.5) * 2;
                }
            }

            draw() {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.strokeStyle = 'rgba(0,0,0,0.3)';
                ctx.stroke();
                ctx.fillStyle = '#000';
                ctx.font = 'bold 9px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(this.number, this.x, this.y);
            }
        }

        function generateCustomDraw() {
            const sumMin = parseInt(document.getElementById('sumMin').value) || 100;
            const sumMax = parseInt(document.getElementById('sumMax').value) || 170;
            const consecutiveMode = document.getElementById('consecutiveFilter').value;

            let attempts = 0;
            let validSet = null;

            while (attempts < 2000) {
                attempts++;
                const candidate = generateCandidateSet();
                
                // Check Sum Range
                const sum = candidate.reduce((a, b) => a + b, 0);
                if (sum < sumMin || sum > sumMax) continue;

                // Check Consecutive Rule
                if (consecutiveMode === 'none' && hasAnyConsecutive(candidate)) continue;
                if (consecutiveMode === 'max2' && hasMoreThanTwoConsecutive(candidate)) continue;

                validSet = candidate;
                break;
            }

            const filterLog = document.getElementById('filterLog');

                state.currentCustomSet = validSet;
                display.innerHTML = validSet.map(n => createBallElement(n, 'w-10 h-10 text-base animate-bounce-short')).join('');

        function generateCandidateSet() {
            const set = new Set(Array.from(state.customInclude));
            const availablePool = [];

            // Hot / Cold simulated weights
            const hotNumbers = [3, 12, 17, 26, 33, 38, 44, 7, 14, 21];
            const coldNumbers = [1, 9, 15, 22, 29, 31, 40, 43, 5, 18];

            for (let i = 1; i <= 45; i++) {
                if (!state.customExclude.has(i) && !set.has(i)) {
                    let weight = 1;
                    if (state.aiMode === 'hot' && hotNumbers.includes(i)) weight = 3;
                    if (state.aiMode === 'cold' && coldNumbers.includes(i)) weight = 3;
                    if (state.aiMode === 'golden' && (i % 2 !== 0)) weight = 1.5;

                    for (let w = 0; w < weight; w++) availablePool.push(i);
                }
            }

            while (set.size < 6 && availablePool.length > 0) {
                const idx = Math.floor(Math.random() * availablePool.length);
                const picked = availablePool[idx];
                set.add(picked);
            }

            return Array.from(set).sort((a, b) => a - b);
        }

        function hasAnyConsecutive(arr) {
            for (let i = 0; i < arr.length - 1; i++) {
                if (arr[i + 1] === arr[i] + 1) return true;
            }
            return false;
        }

        function hasMoreThanTwoConsecutive(arr) {
            for (let i = 0; i < arr.length - 2; i++) {
                if (arr[i + 1] === arr[i] + 1 && arr[i + 2] === arr[i] + 2) return true;
            }
            return false;
        }

            while (drawn.length < 7) {
                const idx = Math.floor(Math.random() * pool.length);
                drawn.push(pool.splice(idx, 1)[0]);
            }
            state.targetWinning = drawn.slice(0, 6).sort((a, b) => a - b);
            state.targetBonus = drawn[6];
            renderTargetWinningUI();
        }

        function runWinningSimulation() {
            if (state.savedHistory.length === 0) {
                return alert('보관함에 저장된 번호가 없습니다. 먼저 번호를 추첨하여 보관함에 저장해 주세요.');
            }

            let rankCount = { '1등': 0, '2등': 0, '3등': 0, '4등': 0, '5등': 0, '낙첨': 0 };

            state.savedHistory.forEach(item => {
                const matchCount = item.numbers.filter(n => state.targetWinning.includes(n)).length;
                const matchBonus = item.numbers.includes(state.targetBonus);

                if (matchCount === 6) rankCount['1등']++;
                else if (matchCount === 5 && matchBonus) rankCount['2등']++;
                else if (matchCount === 5) rankCount['3등']++;
                else if (matchCount === 4) rankCount['4등']++;
                else if (matchCount === 3) rankCount['5등']++;
                else rankCount['낙첨']++;
            });

            const resultBox = document.getElementById('simulationResultBox');
            const stats = document.getElementById('simResultStats');
            resultBox.classList.remove('hidden');

            stats.innerHTML = `
                <div class="flex justify-between"><span>🥇 1등 (6개 일치):</span> <strong class="text-amber-400">${rankCount['1등']}건</strong></div>
                <div class="flex justify-between"><span>🥈 2등 (5개 + 보너스):</span> <strong class="text-slate-200">${rankCount['2등']}건</strong></div>
                <div class="flex justify-between"><span>🥉 3등 (5개 일치):</span> <strong class="text-slate-200">${rankCount['3등']}건</strong></div>
                <div class="flex justify-between"><span>4등 (4개 일치):</span> <span>${rankCount['4등']}건</span></div>
                <div class="flex justify-between"><span>5등 (3개 일치):</span> <span>${rankCount['5등']}건</span></div>
                <div class="flex justify-between text-slate-500"><span>낙첨:</span> <span>${rankCount['낙첨']}건</span></div>
            `;

            playTaDaSound();
        }

        /* Initial Setup On Load */
        window.onload = function() {
            initDrumBalls();
            animateDrum();
            updateFilterGrids();
            loadHistory();
            renderTargetWinningUI();
        };
